function Button({ children, variant = 'primary', loading, ...props }) {
  const variantClass = {
    primary: 'btn-primary',
    success: 'btn-success',
    danger: 'btn-danger',
    secondary: 'btn-secondary',
  }[variant]

  return (
    <button 
      className={`btn ${variantClass}`} 
      disabled={loading || props.disabled}
      {...props}
    >
      {loading ? 'Загрузка...' : children}
    </button>
  )
}

import { useState } from 'react'
import { useAuth } from '../context/AuthContext'
import Input from './ui/Input'
import Button from './ui/Button'

function VerifyEmail({ email, tempToken, onBack, onVerified }) {
  const { verify } = useAuth()
  const [code, setCode] = useState('')
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  const onSubmit = async (e) => {
    e.preventDefault()
    setError('')
    setLoading(true)

    try {
      await verify(tempToken, code)
      onVerified()
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="container">
      <h2>Подтверждение email</h2>
      <p>
        Код отправлен на: <strong>{email}</strong>
      </p>

      <form onSubmit={onSubmit} autoComplete="off">
        <Input
          label="Код подтверждения"
          type="text"
          value={code}
          onChange={(e) => setCode(e.target.value)}
          required
          placeholder="Введите код из письма"
        />

        {error && <div className="error">{error}</div>}

        <Button type="submit" loading={loading}>
          Подтвердить
        </Button>

        <Button type="button" variant="secondary" onClick={onBack}>
          Назад
        </Button>
      </form>
    </div>
  )
}

import { useState } from 'react'
import { useAuth } from '../context/AuthContext'
import Input from './ui/Input'
import Button from './ui/Button'

function Register({ onSwitchToLogin, onRegistered }) {
  const { register } = useAuth()
  const [values, setValues] = useState({
    email: '',
    phone: '+',
    password: '',
    ifsms: false,
  })
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  const handleChange = (e) => {
    const { name, value, type, checked } = e.target
    setValues({
      ...values,
      [name]: type === 'checkbox' ? checked : value,
    })
    setError('')
  }

  const onSubmit = async (e) => {
    e.preventDefault()
    setError('')
    
    const phoneDigits = values.phone.replace(/\D/g, '')
    if (phoneDigits.length < 9) {
      setError('Телефон должен содержать минимум 9 цифр')
      return
    }
    
    setLoading(true)

    try {
      const result = await register(values.email, values.phone, values.password, values.ifsms)
      console.log('Результат регистрации:', result)
      onRegistered(values.email, result.access_token)
    } catch (err) {
      setError(err.response?.data?.detail || err.message || 'Ошибка регистрации')  // ← ИСПРАВИТЬ
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="container">
      <h2>Регистрация</h2>

      <form onSubmit={onSubmit} autoComplete="off">
        <Input
          label="Email"
          type="email"
          name="email"
          value={values.email}
          onChange={handleChange}
          required
        />

        <Input
          label="Телефон"
          type="tel"
          name="phone"
          value={values.phone}
          onChange={handleChange}
          required
          pattern="\+[0-9]{9,}"
          placeholder="+79991234567"
        />

        <Input
          label="Пароль (минимум 8 символов)"
          type="password"
          name="password"
          value={values.password}
          onChange={handleChange}
          required
          minLength="8"
          autoComplete="new-password"
        />

        <div className="form-group">
          <label className="checkbox-label">
            <input
              type="checkbox"
              name="ifsms"
              checked={values.ifsms}
              onChange={handleChange}
            />
            Получить код подтверждения по SMS (вместо email)
          </label>
        </div>

        {error && <div className="error">{error}</div>}

        <Button type="submit" variant="success" loading={loading}>
          Зарегистрироваться
        </Button>
      </form>

      <div className="text-center">
        <button onClick={onSwitchToLogin} className="link-button">
          Уже есть аккаунт? Войти
        </button>
      </div>
    </div>
  )
}

function Login({ onSwitchToRegister }) {
  const { login } = useAuth()
  const { values, errors, loading, handleChange, handleSubmit } = useForm({
    email: '',
    password: '',
  })

  const onSubmit = async (e) => {
    e.preventDefault()
    setErrors({})
    setLoading(true)

  return (

  const loadTenants = async () => {
    try {
      setLoading(true)
      const data = await getTenants()
      setTenants(data)
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  const handleSelectTenant = async (tenantId, e) => {
    e.stopPropagation() // Останавливаем всплытие, чтобы не сработал onClick карточки
    if (tenantId === currentTenantId) return // Уже активный

    console.log('Switching to tenant:', tenantId) // DEBUG
    
    try {
      setSwitching(tenantId)
      const data = await switchTenant(tenantId)
      console.log('Switch response:', data) // DEBUG
      
      // Обновляем токен через AuthContext
      updateToken(data.access_token, data.refresh_token)
      
      // Уведомляем родительский компонент об изменении активного тенанта
      if (onTenantSelected) {
        onTenantSelected(tenantId)
      }
    } catch (err) {
      setError(err.message)
    } finally {
      setSwitching(null)
    }
  }

  if (loading) {
    return <div className="dashboard">Загрузка тенантов...</div>
  }

  if (error) {
    return <div className="dashboard error">{error}</div>
  }

  return (
    <div className="dashboard">
      <h2 style={{ marginBottom: '20px' }}>Ваши тенанты</h2>
      {tenants.length === 0 ? (
        <div style={{ textAlign: 'center', padding: '40px' }}>
          <p style={{ marginBottom: '20px', color: '#666' }}>
            У вас пока нет ни одного тенанта
          </p>
          <Button variant="success" onClick={onCreateClick}>
            Создать первый тенант
          </Button>
        </div>
      ) : (
        <>
          <div style={{ marginBottom: '20px' }}>
            {tenants.map((tenant) => (
              <div
                key={tenant.id}
                className={`tenant-card ${tenant.id === currentTenantId ? 'active' : ''}`}
                onClick={() => onTenantClick(tenant)}
                style={{ 
                  cursor: 'pointer',
                  display: 'flex',
                  justifyContent: 'space-between',
                  alignItems: 'center'
                }}
              >
                <div>
                  <h4>
                    {tenant.name}
                    {tenant.id === currentTenantId && (
                      <span className="tenant-badge">АКТИВНЫЙ</span>
                    )}
                  </h4>
                  <p>
                    Локаль: {tenant.locale} • Оплата: {tenant.payment} • Статус: {tenant.status}
                  </p>
                </div>
                
                {/* Кнопка активации тенанта - показываем только для неактивных */}
                {tenant.id !== currentTenantId && (
                  <Button
                    variant="primary"
                    onClick={(e) => handleSelectTenant(tenant.id, e)}
                    disabled={switching === tenant.id}
                    style={{ whiteSpace: 'nowrap', padding: '8px 16px' }}
                  >
                    {switching === tenant.id ? 'Переключение...' : 'Активировать'}
                  </Button>
                )}

            ))}
          </div>
          <Button variant="success" onClick={onCreateClick}>
            + Создать новый тенант
          </Button>
        </>
      )}

import { useState } from 'react'

export const useForm = (initialValues) => {
  const [values, setValues] = useState(initialValues)
  const [errors, setErrors] = useState({})
  const [loading, setLoading] = useState(false)

  const handleChange = (e) => {
    const { name, value, type, checked } = e.target
    setValues({
      ...values,
      [name]: type === 'checkbox' ? checked : value,
    })
    // Очистить ошибку при изменении поля
    if (errors[name]) {
      setErrors({ ...errors, [name]: '' })
    }
  }

  const handleSubmit = async (callback, validate) => {
    return async (e) => {
      e.preventDefault()
      setErrors({})

      // Валидация если есть
      if (validate) {
        const validationErrors = validate(values)
        if (Object.keys(validationErrors).length > 0) {
          setErrors(validationErrors)
          return
        }
      }

      setLoading(true)
      try {
        await callback(values)
      } catch (error) {
        setErrors({ submit: error.message })
      } finally {
        setLoading(false)
      }
    }
  }

  return {
    values,
    errors,
    loading,
    handleChange,
    handleSubmit,
    setLoading,
    setErrors,
  }

export const verify = async (token, code) => {
  // Для verify используем временный токен, не из localStorage
  const response = await apiClient.post(
    `/api/v1/auth/verify?code=${code}`,
    {},
    {
      headers: {
        Authorization: `Bearer ${token}`,
      },
    }
  )
  return response.data
}

export const login = async (username, password) => {
  // Login использует form-urlencoded, поэтому fetch
  const formData = new URLSearchParams()
  formData.append('username', username)
  formData.append('password', password)

  const response = await fetch(`${API_URL}/api/v1/auth/login`, {
    method: 'POST',
    headers: {
      accept: 'application/json',
      'Content-Type': 'application/x-www-form-urlencoded',
    },
    body: formData,
  })

  if (!response.ok) {
    const error = await response.json()
    throw new Error(error.detail || 'Ошибка входа')
  }

  return await response.json()

import { createContext, useContext, useState, useEffect } from 'react'
import { login as apiLogin, register as apiRegister, verify as apiVerify } from '../api/auth'

const AuthContext = createContext(null)

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null)
  const [token, setToken] = useState(null)
  const [loading, setLoading] = useState(true)

  // Инициализация - проверяем есть ли токен в localStorage
  useEffect(() => {
    const savedToken = localStorage.getItem('token')
    const savedEmail = localStorage.getItem('user_email')
    
    if (savedToken) {
      setToken(savedToken)
      setUser(savedEmail ? { email: savedEmail } : null)
    }
    setLoading(false)
  }, [])

  const login = async (email, password) => {
    const data = await apiLogin(email, password)
    setToken(data.access_token)
    setUser({ email })
    localStorage.setItem('token', data.access_token)
    localStorage.setItem('refresh_token', data.refresh_token)
    localStorage.setItem('user_email', email)
    return data
  }

  const register = async (email, phone, password, ifsms) => {
    const data = await apiRegister(email, phone, password, ifsms)
    return data // Возвращаем токен для верификации
  }

  const verify = async (tempToken, code) => {
    const data = await apiVerify(tempToken, code)
    return data
  }

  // Новая функция для обновления токена (например, после switch-tenant)
  const updateToken = (newToken, newRefreshToken) => {
    setToken(newToken)
    localStorage.setItem('token', newToken)
    if (newRefreshToken) {
      localStorage.setItem('refresh_token', newRefreshToken)
    }
  }

  const logout = () => {
    setToken(null)
    setUser(null)
    localStorage.removeItem('token')
    localStorage.removeItem('refresh_token')
    localStorage.removeItem('user_email')
  }

  const value = {
    user,
    token,
    loading,
    login,
    register,
    verify,
    logout,
    updateToken,
    isAuthenticated: !!token,
  }

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>
}

  const context = useContext(AuthContext)
  if (!context) {
    throw new Error('useAuth должен использоваться внутри AuthProvider')
  }

import { useState, useEffect } from 'react'
import { useAuth } from './context/AuthContext'
import Login from './components/Login'
import Register from './components/Register'
import VerifyEmail from './components/VerifyEmail'
import Header from './components/Header'
import TenantList from './components/TenantList'
import TenantDetails from './components/TenantDetails'
import CreateTenant from './components/CreateTenant'
import { getTenant } from './api/tenants'
import EditTenant from './components/EditTenant'
import BankingDetailsRu from './components/BankingDetailsRu'


function App() {
  const { user, isAuthenticated, logout, loading } = useAuth()
  const [view, setView] = useState('login')
  const [pendingEmail, setPendingEmail] = useState(null)
  const [pendingToken, setPendingToken] = useState(null)
  const [currentTenant, setCurrentTenant] = useState(null)
  const [editingTenant, setEditingTenant] = useState(null)

  useEffect(() => {
    if (isAuthenticated) {
      const token = localStorage.getItem('token')
      if (token) {
        try {
          const payload = JSON.parse(atob(token.split('.')[1]))
          if (payload.tenant_id) {
            setCurrentTenant({ id: payload.tenant_id })
          }
        } catch (e) {
          console.error('Ошибка декодирования токена:', e)
        }
      }
      setView('tenants')
    }
  }, [isAuthenticated])

  if (loading) {
    return <div className="dashboard">Загрузка...</div>
  }

  // === AUTH VIEWS ===
  if (!isAuthenticated) {
    const handleRegistered = (email, token) => {
      setPendingEmail(email)
      setPendingToken(token)
      setView('verify')
    }

    const handleVerified = () => {
      alert('Email подтверждён! Теперь можете войти.')
      setPendingEmail(null)
      setPendingToken(null)
      setView('login')
    }

    const handleBackToLogin = () => {
      setPendingEmail(null)
      setPendingToken(null)
      setView('login')
    }

    if (view === 'verify') {
      return (
        <VerifyEmail
          email={pendingEmail}
          tempToken={pendingToken}
          onBack={handleBackToLogin}
          onVerified={handleVerified}
        />
      )
    }

    if (view === 'register') {
      return (
        <Register
          onSwitchToLogin={() => setView('login')}
          onRegistered={handleRegistered}
        />
      )
    }

    return <Login onSwitchToRegister={() => setView('register')} />
  }

  // === HANDLERS ===
  const handleNavigate = (section) => {
    setView(section)
    if (section === 'tenants') {
      setEditingTenant(null)
    }
  }

  const handleTenantSelected = async (tenantId) => {
    try {
      const tenantData = await getTenant(tenantId)
      setCurrentTenant(tenantData)
    } catch (err) {
      setCurrentTenant({ id: tenantId })
    }
  }

  const handleTenantClick = (tenant) => {
    setEditingTenant(tenant)
    setView('view-tenant')
  }

  const handleEditTenant = (tenant, tab = 'main') => {
    setEditingTenant({ ...tenant, tab })
    setView('edit-tenant')
  }

  const handleTenantCreated = () => {
    setView('tenants')
  }

  // === MAIN APP ===
  return (
    <>
      <Header currentTenant={currentTenant} onNavigate={handleNavigate} />
      
      {view === 'create-tenant' && (
        <CreateTenant
          onCreated={handleTenantCreated}
          onCancel={() => setView('tenants')}
        />
      )}

      {view === 'tenants' && (
        <TenantList
          currentTenantId={currentTenant?.id}
          onTenantClick={handleTenantClick}
          onCreateClick={() => setView('create-tenant')}
          onTenantSelected={handleTenantSelected}
        />
      )}

      {view === 'view-tenant' && editingTenant && (
        <TenantDetails
          tenant={editingTenant}
          onEdit={() => {
            setView('edit-tenant')
            setEditingTenant({ ...editingTenant, tab: 'main' })
          }}
          onEditBanking={() => {
            setView('edit-tenant')
            setEditingTenant({ ...editingTenant, tab: 'banking' })
          }}
          onBack={() => setView('tenants')}
        />
      )}


      {view === 'edit-tenant' && editingTenant && (
        <>
          {editingTenant.tab === 'banking' ? (
            <BankingDetailsRu
              tenant={editingTenant}
              onSaved={() => setView('tenants')}
              onCancel={() => setView('tenants')}
            />
          ) : (
            <EditTenant
              tenant={editingTenant}
              onSaved={() => setView('tenants')}
              onCancel={() => setView('tenants')}
            />
          )}
        </>
      )}


      {view === 'lines' && (
        <div className="dashboard">
          <h2>Линии мессенджеров</h2>
          <p>TODO: Раздел линий</p>
        </div>

        <div className="dashboard">

          <p>TODO: Раздел биллинга</p>
        </div>

        <div className="dashboard">
          <h2>Тарифы</h2>
          <p>TODO: Раздел тарифов</p>
        </div>
      )}
